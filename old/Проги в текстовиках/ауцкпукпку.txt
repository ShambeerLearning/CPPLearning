//Минимум максимум в массиве
void MinMax(int *Mass, int rows, int &Minimum, int &Maximum)
{

    Minimum = Mass[0];
    Maximum = Mass[0];

    for(int i = 0; i < rows; i++)
    {
        if(Mass[i]<0 && Mass[i]<Minimum)
        {
            Minimum = Mass[i];
        }
        if(Mass[i]> 0 && Mass[i]>Maximum)
        {
            Maximum = Mass[i];
        }
    }


}

//Вывод элементов массива на экран
void print(int* Mass, int rows)
{
    for(int i = 0; i < rows; i++)
        {
            cout << Mass[i] << " ";
        }
}

//Вывод элементов матрицы на экран
void print(int** Matr, int rows, int column)
{
    for(int i = 0; i < rows; i++)
        {
            for(int j = 0; j < column; j++)
                {
                    cout << Matr[i][j] << " ";
                }
            cout << endl;
        }
}

void zadachaOne(int *Mass, int rows)
{
    int Minimum, Maximum;
    MinMax(Mass,rows,Minimum,Maximum);

    for(int i = 0; i < rows; i++)
    {
        if(Mass[i] > Minimum && Mass[i] < 0)
            {
                Minimum = Mass[i];
            }
        if(Mass[i] < Maximum && Mass[i] > 0)
            {
                Maximum = Mass[i];
            }
    }

    cout << "Динамический массив\n";
    print(Mass,rows);
    cout << "\nМаксимальный минимум и Минимальный максимум\n";
    cout << '\n' << Minimum << " " << Maximum;

}

void zadachaTree(int **Matr, int rows, int column)
{
    int n = rows * column;
    cout << "элементы в матрице заполняются" << n << endl;
    //начальная строка
    int m0 = 0;
    //инкремент или декремент строки
    int up = 1;
    //флаг заполнения: заполнять столбец вверх или вниз
    Matr[m0][0] = 1;
    //заполняем начальную строку
    for (int i = 1; i < column; i++)
        {
            if (i % 2 == 0)
                {
                    Matr[m0][i] = i * rows + 1;
                }
            else
                {
                  Matr[m0][i] = rows * (i + 1);
                }
        }
    //заполняем остальные строки (чётный столбец = значение в предыдущей строке + 1),
    //нечётный = зн. в пред. строке + 1
    for (int m1 = m0 + up; m1 >= 0 && m1 < rows; m1 += up)
    {
        for (int i = 0; i < column; i++)
            {
                if (i % 2 == 0)
                {
                    Matr[m1][i] = Matr[m1 - up][i] + 1;
                }
                else
                {
                    Matr[m1][i] = Matr[m1 - up][i] - 1;
                }
            }
    }

}
