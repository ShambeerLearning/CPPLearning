#include <iostream>
#include <cmath>
int main()
{
	setlocale(LC_ALL, "ru");
	std::cout << "Введите х и у - ";
	double x2, y2, f, x1 = 0, y1 = 0;
	std::cin >> y2 >> x2;
	const double rad = 9.00;
	const double delta = 0.5;
	f = pow((x2 - x1), 2) + pow((y2 - y1), 2);
	if (abs(f-rad) < delta)
	{
		std::cout << "Принадлежит" << std::endl;
	}
	else
	{
		std::cout << "Не принадлежит" << std::endl;
	}
	system("pause");
}
