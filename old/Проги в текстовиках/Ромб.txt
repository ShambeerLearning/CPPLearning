#include "stdafx.h"
#include <iostream>

int main()
{
	char b;
	setlocale(LC_ALL, "ru");
	switch (b)
	{
	case '*':
		std::cout << "*";
		break;
	case '+':
		std::cout << "+";
		break;
	case '#':
		std::cout << "#";
		break;
	default:
		std::cout << "Ошибка, введите корректный символ";

	}
	std::system("pause");
}
